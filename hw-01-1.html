# Імпортуємо обєкт date з модуля datetime from datetime import date def get_days_from_today(other_date: str) -> int: """ Функція повертає ціле число, яке вказує на кількість днів від заданої дати до поточної. На вході: other_date: дата у форматі 'YYYY-MM-DD' На виході: повертає ціле число, яке вказує на кількість днів від заданої дати до поточної. Якщо задана дата пізніша за поточну, результат має бути від'ємним. """ today = date.today() # метод поточна дата other = date.fromisoformat(other_date) # метод конвертує строчку формату YYYY-MM-DD у дату delta = today - other # різниця дат return delta.days # метод days вертає дні # ініціалізація кількості спроб i = 1 while i <= 3: # даємо 3 спроби для правильного введення дати # введення строчки YYYY-MM-DD користувачем input_date_string = input("Задайте дату у форматі РРРР-ММ-ДД:").strip() # метод видаляє лідируючі і кінцеві пробіли try: diff = get_days_from_today(input_date_string) # отримуємо ціле число днів print(diff) # виводимо на термінал i = 4 # умова для виходу з циклу, коли нема виключення except ValueError: # виключення print("Неправильний формат.") i = i + 1 # пеходимо на наступну спробу if i >= 3: # кількість спроб перевищено print("Було надано 3 спроби введення коректних даних.")
